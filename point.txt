function buildWallMeshData(line, baseZ, height, wallId) {
        const [[lon1, lat1], [lon2, lat2]] = line;

        const z1 = baseZ;
        const z2 = baseZ + height;

        const positions = [
            lon1, lat1, z1,
            lon2, lat2, z1,
            lon2, lat2, z2,
            lon1, lat1, z2
        ];

        const uvs = [
            0, 0,
            1, 0,
            1, 1,
            0, 1
        ];

        const faces = [
            0, 1, 2,
            0, 2, 3
        ];

        // metadata để gắn object sau này
        const wallMeta = {
            id: wallId,
            p0: [lon1, lat1],
            p1: [lon2, lat2],
            baseZ,
            height
        };

        return { positions, uvs, faces, wallMeta };
    }

    function buildHighlightWallMesh(wall) {
        const { p0, p1, baseZ, height } = wall;

        return new Mesh({
            spatialReference: { wkid: 4326 },
            vertexAttributes: {
                position: [
                    p0[0], p0[1], baseZ,
                    p1[0], p1[1], baseZ,
                    p1[0], p1[1], baseZ + height,
                    p0[0], p0[1], baseZ + height
                ]
            },
            components: [{
                faces: [
                    0, 1, 2,
                    0, 2, 3
                ]
            }]
        });
    }


    const highlightLayer = new GraphicsLayer();
    map.add(highlightLayer);
    
    // Tạo một GeoJSONLayer từ file hoặc URL
    fetch("./tuong.geojson")
        .then(res => res.json())
        .then(data => {

            const wallsByBuilding = {};

            data.features.forEach(f => {
                const bId = f.properties.buildingId;
                if (!wallsByBuilding[bId]) wallsByBuilding[bId] = [];
                wallsByBuilding[bId].push(f);
            });

            Object.values(wallsByBuilding).forEach(walls => {

                let positions = [];
                let uvs = [];
                let faces = [];
                let vertexOffset = 0;
                let faceOffset = 0;

                const wallIndex = {};

                walls.forEach(f => {
                    const { baseZ, height, id } = f.properties;
                    const line = f.geometry.coordinates;

                    const wall = buildWallMeshData(line, baseZ, height, id);

                    positions.push(...wall.positions);
                    uvs.push(...wall.uvs);

                    wall.faces.forEach(i => faces.push(i + vertexOffset));

                    // 🔑 map faceIndex → wall
                    wallIndex[id] = {
                        ...wall.wallMeta,
                        faceStart: faceOffset,
                        faceEnd: faceOffset + 1
                    };

                    vertexOffset += 4;
                    faceOffset += 2;
                });

                const mesh = new Mesh({
                    spatialReference: { wkid: 4326 },
                    vertexAttributes: {
                        position: positions,
                        uv: uvs
                    },
                    components: [{
                        faces
                    }]
                });

                const buildingGraphic = new Graphic({
                    geometry: mesh,
                    symbol: {
                        type: "mesh-3d",
                        symbolLayers: [{
                            type: "fill",
                            material: {
                                texture: {
                                    url: "./Images/Walls/1_Tuong_gach.jpg"
                                }
                            }
                        }]
                    },
                    attributes: {
                        buildingId: walls[0].properties.buildingId,
                        buildingName: walls[0].properties.buildingName
                    }
                });

                // 🔥 gắn index wall vào GRAPHIC
                buildingGraphic.wallIndex = wallIndex;

                graphicsLayer.add(buildingGraphic);
            });
        });

    view.on("click", async (event) => {

        const response = await view.hitTest(event);

        const hit = response.results.find(r =>
            r.graphic?.wallIndex &&
            typeof r.faceIndex === "number"
        );

        if (!hit) return;

        const { graphic, faceIndex } = hit;
        const wallIndex = graphic.wallIndex;

        let pickedWall = null;

        for (const id in wallIndex) {
            const w = wallIndex[id];
            if (faceIndex >= w.faceStart && faceIndex <= w.faceEnd) {
                pickedWall = w;
                break;
            }
        }

        if (!pickedWall) return;

        // 🧹 clear highlight cũ
        highlightLayer.removeAll();

        // 🧱 tạo highlight overlay
        const highlightGraphic = new Graphic({
            geometry: buildHighlightWallMesh(pickedWall),
            symbol: {
                type: "mesh-3d",
                symbolLayers: [{
                    type: "fill",
                    material: {
                        color: [255, 215, 0, 0.6]
                    },
                    edges: {
                        type: "solid",
                        color: "yellow",
                        size: 2
                    }
                }]
            }
        });

        highlightLayer.add(highlightGraphic);

        // 📋 popup wall
        view.popup.open({
            title: "Thông tin tường",
            content: `
            <b>ID:</b> ${pickedWall.id}<br>
            <b>Chiều cao:</b> ${pickedWall.height} m<br>
            <b>Base Z:</b> ${pickedWall.baseZ}
        `,
            location: event.mapPoint
        });
    });
